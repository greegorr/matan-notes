\definitionticket{11}{Ячейка}
\label{definition:11}

\begin{definition}
Пусть $a,b\in\mathbb R^m$, $a_i\le b_i$ для всех $i$.
\textbf{Ячейкой} в $\mathbb R^m$ называется множество
\[
  [a,b):=\prod_{i=1}^{m}[a_i,b_i)
  =\{x\in\mathbb R^m:\ a_i\le x_i<b_i,\ i=1,\ldots,m\}.
\]
\end{definition}
