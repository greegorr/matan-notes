\definitionticket{23}{Интеграл ступенчатой функции}
\label{definition:23}

\begin{definition}
Пусть $(X,\mathcal A,\mu)$ --- пространство с мерой, а $f$ --- неотрицательная ступенчатая функция с допустимым разбиением
\[
  X=\bigsqcup_{k=1}^{n}E_k,
  \qquad E_k\in\mathcal A,\quad f|_{E_k}=c_k\ge0.
\]
Её \textbf{интегралом} называется
\[
  \int_X f\,d\mu:=\sum_{k=1}^{n}c_k\,\mu(E_k)\in[0,+\infty].
\]
Здесь принято соглашение $0\cdot(+\infty)=0$.
\end{definition}
