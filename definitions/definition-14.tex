\definitionticket{14}{Формулировка теоремы о непрерывности меры снизу}
\label{definition:14}

\begin{theorem}

\emph{1 (о непрерывности меры снизу.} Пусть \(\mathcal{A}\) --- σ-алгебра, \(\mu:\mathcal{A}\to\overline{\mathbb{R}}\) --- объём. Эквивалентны:

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \(\mu\) --- мера (счётно аддитивна);
\item
  \(\mu\) \textbf{непрерывна снизу}: для любых \(A_n,A\in\mathcal{A}\) \[A_1\subset A_2\subset\cdots,\qquad A=\bigcup_{n=1}^\infty A_n \;\Rightarrow\; \mu A=\lim_{n\to\infty}\mu A_n.\]
\end{enumerate}

\end{theorem}
