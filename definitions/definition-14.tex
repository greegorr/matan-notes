\definitionticket{14}{Формулировка теоремы о непрерывности меры снизу}
\label{definition:14}

\begin{theorem}
Пусть $\mathcal A$ --- $\sigma$-алгебра, а $\mu:\mathcal A\to[0,+\infty]$ --- объём.
Эквивалентны следующие утверждения:
\begin{enumerate}
  \item $\mu$ --- мера;
  \item $\mu$ \textbf{непрерывна снизу}: для любых $A_n\in\mathcal A$, $A_1\subset A_2\subset\cdots$, выполнено
  \[
    \mu\left(\bigcup_{n=1}^{\infty}A_n\right)
    =\lim_{n\to\infty}\mu(A_n).
  \]
\end{enumerate}
\end{theorem}
