\definitionticket{24}{Интеграл неотрицательной измеримой функции}
\label{definition:24}

\begin{definition}
Пусть $(X,\mathcal A,\mu)$ --- пространство с мерой, а $f:X\to[0,+\infty]$ --- измеримая функция.
Её \textbf{интеграл} определяется равенством
\[
  \int_X f\,d\mu:=
  \sup_{\substack{0\le g\le f\\g\ \text{измеримая ступенчатая}}}
  \int_X g\,d\mu\ \in[0,+\infty].
\]
\end{definition}
