\definitionticket{27}{Измеримое отображение}
\label{definition:27}

\begin{definition}
Пусть $(X,\mathcal A)$ и $(Y,\mathcal B)$ --- множества с $\sigma$-алгебрами.
Отображение $\varphi:X\to Y$ называется \textbf{измеримым}, если
\[
  \varphi^{-1}(E)\in\mathcal A\qquad\text{для всех }E\in\mathcal B.
\]
\end{definition}
