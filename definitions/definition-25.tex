\definitionticket{25}{Суммируемая функция}
\label{definition:25}

\begin{definition}
Пусть $(X,\mathcal A,\mu)$ --- пространство с мерой, $f:X\to\overline{\mathbb R}$ --- измеримая функция и
\[
  f^+:=\max(f,0),\qquad f^-:=\max(-f,0).
\]
Функция $f$ называется \textbf{суммируемой}, если
\[
  \int_X f^+\,d\mu<+\infty
  \quad\text{и}\quad
  \int_X f^-\,d\mu<+\infty.
\]
\end{definition}
