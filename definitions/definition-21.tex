\definitionticket{21}{Измеримая функция}
\label{definition:21}

\begin{definition}
Пусть $(X,\mathcal A,\mu)$ --- пространство с мерой, $E\in\mathcal A$ и
\[
  f:E\to\overline{\mathbb R},
  \qquad \overline{\mathbb R}=\mathbb R\cup\{-\infty,+\infty\}.
\]
Функция $f$ называется \textbf{измеримой на $E$}, если для любого $a\in\mathbb R$
\[
  \{x\in E:\ f(x)<a\}\in\mathcal A.
\]
\end{definition}
