\ticket{02}{Полиномиальная формула}
\label{ticket:02}

\begin{definition}
\textbf{Мультииндекс} --- набор $\alpha=(\alpha_1,\ldots,\alpha_m)\in\mathbb N_0^m$.
Используются обозначения
\[
  |\alpha|=\alpha_1+\cdots+\alpha_m,\quad
  \alpha!=\alpha_1!\cdots\alpha_m!,\quad
  x^\alpha=x_1^{\alpha_1}\cdots x_m^{\alpha_m},
\]
\[
  D^\alpha f=\frac{\partial^{|\alpha|}f}
  {\partial x_1^{\alpha_1}\cdots\partial x_m^{\alpha_m}}.
\]
\end{definition}

\begin{lemma}
Для элементов $a_1,\ldots,a_m$ коммутативной алгебры с единицей и $r\in\mathbb N$ выполнено
\[
  (a_1+\cdots+a_m)^r
  =\sum_{|\alpha|=r}\frac{r!}{\alpha!}a^\alpha.
\]
\end{lemma}

\begin{proof}
При $r=1$ формула очевидна.
Пусть она верна для $r$.
Умножая на $\sum_{s=1}^m a_s$, получаем
\[
  (a_1+\cdots+a_m)^{r+1}
  =\sum_{s=1}^m\sum_{|\alpha|=r}\frac{r!}{\alpha!}a^{\alpha+e_s}.
\]
В $s$-й сумме положим $\beta=\alpha+e_s$.
Тогда $|\beta|=r+1$, $\beta_s\ge1$ и $1/\alpha!=\beta_s/\beta!$.
При $\beta_s=0$ добавленное слагаемое равно нулю, поэтому
\[
  \begin{aligned}
  (a_1+\cdots+a_m)^{r+1}
   &=\sum_{|\beta|=r+1}\frac{r!}{\beta!}
     \left(\sum_{s=1}^m\beta_s\right)a^\beta\\
   &=\sum_{|\beta|=r+1}\frac{(r+1)!}{\beta!}a^\beta.
  \end{aligned}
\]
\end{proof}
